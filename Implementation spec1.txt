backend/manager.py

python
import os
import uuid
import json
from fastapi import FastAPI, HTTPException, UploadFile, File, Form
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import FileResponse
from pydantic import BaseModel
import redis

app = FastAPI(title="Interview Ingestion Manager")

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

cache_client = redis.Redis(host='redis', port=6379, db=0, decode_responses=True)
broker_client = redis.Redis(host='redis', port=6379, db=1)

class CodeMutationPayload(BaseModel):
    session_id: str
    session_timestamp_ms: int
    data: dict

@app.post("/api/session/start")
def start_session(domain_id: str):
    session_id = str(uuid.uuid4())
    cache_client.hset(f"session:{session_id}", mapping={"domain_id": domain_id, "status": "ACTIVE"})
    return {"session_id": session_id, "status": "ACTIVE"}

@app.post("/api/session/code/event")
def receive_code_event(payload: CodeMutationPayload):
    session_meta = cache_client.hgetall(f"session:{payload.session_id}")
    if not session_meta or session_meta.get("status") != "ACTIVE":
        raise HTTPException(status_code=400, detail="Invalid or inactive session identifier.")
    
    event_data = {
        "event_type": "CODE_MUTATION",
        "session_timestamp_ms": payload.session_timestamp_ms,
        "data": payload.data
    }
    cache_client.rpush(f"session:{payload.session_id}:code", json.dumps(event_data))
    return {"status": "ACK"}

@app.post("/api/session/audio/chunk")
async def receive_audio_chunk(
    session_id: str = Form(...),
    timestamp_ms: int = Form(...),
    chunk: UploadFile = File(...)
):
    session_meta = cache_client.hgetall(f"session:{session_id}")
    if not session_meta:
        raise HTTPException(status_code=400, detail="Session context missing.")

    upload_dir = f"/tmp/sessions/{session_id}"
    os.makedirs(upload_dir, exist_ok=True)
    chunk_path = f"{upload_dir}/chunk_{timestamp_ms}.webm"
    
    with open(chunk_path, "wb") as buffer:
        buffer.write(await chunk.read())
        
    cache_client.rpush(f"session:{session_id}:audio_manifest", json.dumps({
        "timestamp_ms": timestamp_ms,
        "file_path": chunk_path
    }))
    return {"status": "ACK"}

@app.post("/api/session/submit")
def submit_session(session_id: str):
    session_meta = cache_client.hgetall(f"session:{session_id}")
    if not session_meta:
        raise HTTPException(status_code=404, detail="Target session not found.")
        
    cache_client.hset(f"session:{session_id}", "status", "PROCESSING")
    
    job_payload = {
        "session_id": session_id,
        "domain_id": session_meta["domain_id"]
    }
    broker_client.rpush("queue:processing_jobs", json.dumps(job_payload))
    return {"status": "PROCESSING"}

@app.get("/api/session/{session_id}/result")
def get_result(session_id: str):
    status = cache_client.hget(f"session:{session_id}", "status")
    if not status:
        raise HTTPException(status_code=404, detail="Session not found")
    if status == "PROCESSING":
        return {"status": "PROCESSING"}
    if status == "FAILED":
        return {"status": "FAILED"}
        
    result_raw = cache_client.get(f"session:{session_id}:result")
    return json.loads(result_raw)

@app.get("/api/session/{session_id}/download-card")
def download_card(session_id: str):
    card_path = f"/tmp/sessions/{session_id}/share_card.png"
    if not os.path.exists(card_path):
        raise HTTPException(status_code=444, detail="Card file missing or compilation incomplete.")
    return FileResponse(card_path, media_type="image/png", filename="interview_scorecard.png")

import time
import json
import os
import subprocess
import redis
from typing import List, Dict
from exporter import generate_social_share_card

cache_client = redis.Redis(host='redis', port=6379, db=0, decode_responses=True)
broker_client = redis.Redis(host='redis', port=6379, db=1)

def run_audio_stitching_and_transcription(session_id: str) -> List[Dict]:
    manifest_items = cache_client.lrange(f"session:{session_id}:audio_manifest", 0, -1)
    upload_dir = f"/tmp/sessions/{session_id}"
    list_file_path = f"{upload_dir}/concat_list.txt"
    output_audio_path = f"{upload_dir}/stitched.wav"
    
    # 1. Parse chunk manifest and order files chronologically
    chunks = sorted([json.loads(item) for item in manifest_items], key=lambda x: x["timestamp_ms"])
    
    if not chunks:
        return []

    # 2. Native FFmpeg Concatenation Execution Pass
    with open(list_file_path, "w") as f:
        for chunk in chunks:
            f.write(f"file '{chunk['file_path']}'\n")
            
    cmd = ["ffmpeg", "-y", "-f", "concat", "-safe", "0", "-i", list_file_path, "-ar", "16000", "-ac", "1", output_audio_path]
    subprocess.run(cmd, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, check=True)
    
    # Mocking standard word-level token payload shape returned downstream by tracking tools
    base_time = chunks[0]["timestamp_ms"]
    return [
        {"session_timestamp_ms": base_time + 1000, "word": "let's", "confidence": 0.98},
        {"session_timestamp_ms": base_time + 1500, "word": "initialize", "confidence": 0.99},
        {"session_timestamp_ms": base_time + 2000, "word": "the", "confidence": 0.97},
        {"session_timestamp_ms": base_time + 2500, "word": "data", "confidence": 0.99},
        {"session_timestamp_ms": base_time + 3000, "word": "matrix", "confidence": 0.95}
    ]

def run_timeline_alignment_algorithm(code_events: List[Dict], speech_tokens: List[Dict]) -> List[Dict]:
    for event in code_events:
        event["event_type"] = "CODE_MUTATION"
    for token in speech_tokens:
        token["event_type"] = "SPEECH_TOKEN"
        
    unified_timeline = sorted(code_events + speech_tokens, key=lambda x: x["session_timestamp_ms"])
    
    last_speech_ms = 0
    for block in unified_timeline:
        if block["event_type"] == "SPEECH_TOKEN":
            last_speech_ms = block["session_timestamp_ms"]
        elif block["event_type"] == "CODE_MUTATION":
            gap = block["session_timestamp_ms"] - last_speech_ms
            if gap > 12000:
                block["silence_warning"] = True
                block["gap_duration_ms"] = gap
                
    return unified_timeline

def evaluate_via_llm_agent(domain_id: str, aligned_timeline: List[Dict]) -> Dict:
    # Deterministic output simulation matching target evaluation prompt templates
    return {
        "communication_score": 88,
        "structural_classification": "STRATEGIC_THINKER",
        "critique_summary": "Stated explicit architectural assumptions clearly before function implementation. Controlled silence durations effectively."
    }

def start_worker_loop():
    print("[Worker Pool Cluster] Workers actively waiting for jobs...")
    while True:
        job_raw = broker_client.blpop("queue:processing_jobs", timeout=5)
        if not job_raw:
            continue
            
        job_data = json.loads(job_raw[1].decode('utf-8'))
        session_id = job_data["session_id"]
        domain_id = job_data["domain_id"]
        
        try:
            raw_code_logs = cache_client.lrange(f"session:{session_id}:code", 0, -1)
            code_events = [json.loads(x) for x in raw_code_logs]
            
            speech_tokens = run_audio_stitching_and_transcription(session_id)
            aligned_timeline = run_timeline_alignment_algorithm(code_events, speech_tokens)
            final_report = evaluate_via_llm_agent(domain_id, aligned_timeline)
            
            generate_social_share_card(
                session_id=session_id,
                score=final_report["communication_score"],
                persona=final_report["structural_classification"],
                summary=final_report["critique_summary"]
            )
            
            cache_client.hset(f"session:{session_id}", "status", "COMPLETED")
            cache_client.set(f"session:{session_id}:result", json.dumps({
                "status": "COMPLETED",
                "report": final_report,
                "timeline": aligned_timeline
            }))
            
        except Exception as err:
            cache_client.hset(f"session:{session_id}", "status", "FAILED")

if __name__ == "__main__":
    start_worker_loop()

export class AudioStreamer {
  constructor(sessionId, getElapsedMs) {
    this.sessionId = sessionId;
    this.getElapsedMs = getElapsedMs;
    this.mediaRecorder = null;
    this.stream = null;
  }

  async start() {
    this.stream = await navigator.mediaDevices.getUserMedia({ audio: true });
    const audioCtx = new (window.AudioContext || window.webkitAudioContext)({ sampleRate: 16000 });
    const source = audioCtx.createMediaStreamSource(this.stream);
    const destination = audioCtx.createMediaStreamDestination();
    source.connect(destination);

    this.mediaRecorder = new MediaRecorder(destination.stream, { mimeType: 'audio/webm' });
    
    this.mediaRecorder.ondataavailable = async (e) => {
      if (e.data.size > 0) {
        const formData = new FormData();
        formData.append("session_id", this.sessionId);
        formData.append("timestamp_ms", this.getElapsedMs());
        formData.append("chunk", e.data, "chunk.webm");

        fetch("http://localhost:8000/api/session/audio/chunk", {
          method: "POST",
          body: formData
        }).catch(err => console.error("Audio block link drop failure:", err));
      }
    };

    this.mediaRecorder.start(1000);
  }

  stop() {
    if (this.mediaRecorder) this.mediaRecorder.stop();
    if (this.stream) this.stream.getTracks().forEach(track => track.stop());
  }
}

export class CodeTracker {
  constructor(sessionId, getElapsedMs) {
    this.sessionId = sessionId;
    this.getElapsedMs = getElapsedMs;
  }

  attachEditor(editor) {
    editor.onDidChangeModelContent((event) => {
      const payload = {
        session_id: this.sessionId,
        session_timestamp_ms: this.getElapsedMs(),
        data: {
          text_inserted: event.changes[0]?.text || '',
          change_range: event.changes[0]?.range || null,
          total_lines: editor.getModel().getLineCount()
        }
      };

      fetch("http://localhost:8000/api/session/code/event", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload)
      }).catch(err => console.error("Mutation tracing drop failure:", err));
    });
  }
}

import React, { useState, useEffect } from 'react';
import { LineChart, Line, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer } from 'recharts';

export default function AnalyticsDashboard({ sessionId }) {
  const [status, setStatus] = useState('PROCESSING');
  const [data, setData] = useState(null);

  useEffect(() => {
    const interval = setInterval(async () => {
      try {
        const res = await fetch(`http://localhost:8000/api/session/${sessionId}/result`);
        const json = await res.json();
        if (json.status === 'COMPLETED') {
          setData(json);
          setStatus('COMPLETED');
          clearInterval(interval);
        }
      } catch (e) {
        console.error("Poller runtime execution error:", e);
      }

import React, { useContext, useState, useRef } from 'react';
import MonacoEditor from '@monaco-editor/react';
import { UserSessionContext } from './components/UserSessionContext';
import { AudioStreamer } from './components/AudioStreamer';
import { CodeTracker } from './components/CodeTracker';
import AnalyticsDashboard from './components/AnalyticsDashboard';

export default function App() {
  const { sessionState, setSessionState, sessionId, setSessionId, selectedDomain, setSelectedDomain, startClock, getElapsedMs } = useContext(UserSessionContext);
  const [code, setCode] = useState('// Write logic stream components here\n');
  const streamerRef = useRef(null);

  const handleStart = async () => {
    const res = await fetch(`http://localhost:8000/api/session/start?domain_id=${selectedDomain}`, { method: 'POST' });
    const json = await res.json();
    setSessionId(json.session_id);
    startClock();
    
    streamerRef.current = new AudioStreamer(json.session_id, getElapsedMs);
    await streamerRef.current.start();
    setSessionState('ACTIVE');
  };

  const handleEditorMount = (editor) => {
    if (sessionState === 'ACTIVE' && sessionId) {
      const tracker = new CodeTracker(sessionId, getElapsedMs);
      tracker.attachEditor(editor);
    }
  };

  const handleSubmit = async () => {
    streamerRef.current.stop();
    await fetch(`http://localhost:8000/api/session/submit?session_id=${sessionId}`, { method: 'POST' });
    setSessionState('COMPLETED');
  };

  return (
    <div className="min-h-screen bg-gray-100 p-6">
      {sessionState === 'IDLE' && (
        <div className="max-w-md mx-auto bg-white p-6 rounded-xl border border-gray-200 mt-20 text-center space-y-4">
          <h2 className="text-xl font-bold">Select Sandbox Track</h2>
          <select value={selectedDomain} onChange={(e) => setSelectedDomain(e.target.value)} className="w-full p-2 border rounded">
            <option value="backend_systems">Backend Engineering</option>
            <option value="frontend_architecture">Frontend Architecture</option>
          </select>
          <button onClick={handleStart} className="w-full bg-blue-600 text-white font-bold p-3 rounded hover:bg-blue-700 transition">
            Initialize Sandbox Framework
          </button>
        </div>
      )}

      {sessionState === 'ACTIVE' && (
        <div className="grid grid-cols-1 gap-4 max-w-6xl mx-auto">
          <div className="bg-white p-4 rounded-xl border border-gray-200 flex justify-between items-center">
            <div className="flex items-center space-x-2">
              <span className="w-3 h-3 bg-red-500 rounded-full animate-ping" />
              <span className="text-sm font-semibold text-gray-700">Capturing Interleaved Timelines...</span>
            </div>
            <button onClick={handleSubmit} className="bg-emerald-600 text-white px-6 py-2 rounded-lg font-bold hover:bg-emerald-700">
              Submit & Run Grader Pool
            </button>
          </div>
          <div className="h-[500px] bg-white rounded-xl overflow-hidden border border-gray-200 shadow-sm">
            <MonacoEditor height="100%" defaultLanguage="python" value={code} onChange={setCode} onMount={handleEditorMount} />
          </div>
        </div>
      )}

      {sessionState === 'COMPLETED' && <AnalyticsDashboard sessionId={sessionId} />}
    </div>
  );
